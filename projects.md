<script>
function toggleProjCard(btn){
  btn.closest('.proj-card').classList.toggle('open');
}

function openContact(event) {
  if (event) event.preventDefault();
  document.getElementById('contactOverlay').classList.add('open');
  document.body.style.overflow = 'hidden';
}

function closeContact() {
  document.getElementById('contactOverlay').classList.remove('open');
  document.body.style.overflow = '';
}

function sendContact(event) {
  event.preventDefault();
  const name = document.getElementById('contactName').value;
  const email = document.getElementById('contactEmail').value;
  const message = document.getElementById('contactMessage').value;
  const subject = encodeURIComponent('Portfolio contact from ' + name);
  const body = encodeURIComponent(message + '\n\n— ' + name + ' (' + email + ')');
  window.location.href = 'mailto:anandramsathyanarayanan@gmail.com?subject=' + subject + '&body=' + body;
  closeContact();
}

document.addEventListener('keydown', function(e) {
  if (e.key === 'Escape') closeContact();
});
</script>
